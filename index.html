基本構文・制御
// 指定回数の繰り返し (iを変更可能)
for (let i = 0; i < 10; i++) { ... }

// 無限ループ (ゲーム中ずっと実行)
while (true) { ... }

// 条件付きループ
while (条件) { ... }

// 条件分岐
if (条件) { ... }

// 弾の生存期間中に1回だけ実行 (弾側のみ)
once { ... }

// 指定秒数待機する
wait(0.5)

一番上の階層に複数のループを書くと、それらは並行して同時に動きます。

コア (Emitter) & 子弾設定の命令
// 弾を単発発射
spawnBullet("型", "色", 速度, 角度, Xオフセット, Yオフセット, 半径, "画像", "座標モード", 判定半径)

// 弾を全方位(リング状)に発射
spawnRing("型", "色", 速度, 弾数, Xオフセット, Yオフセット, 半径, "画像", "座標モード", 判定半径)

// 弾を扇状(Way)に発射
spawnWay("型", "色", 速度, 角度, 弾数, 拡散角, Xオフセット, Yオフセット, 半径, "画像", "座標モード", 判定半径)

// 子弾設定の魔法陣を生成 (子弾の挙動スクリプトが動きます)
spawnMagicCircle("色", Xオフセット, Yオフセット)

// ターゲットの方向を向く (angle に代入)
aimAtTarget()

// 座標の瞬間移動 / スライド移動
moveTo("位置") または slideTo("位置", 秒数)
※ 位置: "center", "left", "right", "farLeft", "farRight", "leftUp", "rightUp"

// 変数の非同期移行 / 移行完了まで待機
tween("変数名", 開始, 終了, "秒/フレーム/加算/ベクトル等速", 値)
tweenWait("変数名", 開始, 終了, "秒/フレーム/加算/ベクトル等速", 値)
※時間モード: "seconds", "frames", "step", "vecstep" (座標ペア等速用)
弾 (Bullet) の命令
// 速度・角度・色・レーザー幅を変更
speed = 200      // 速度を固定
speed += 10      // 加速
angle += 2       // 曲がる
spriteAngle += 5 // 画像だけ回転 (見た目のみ)
color = #ff3333  // 色変更
laserWidth = 16  // レーザー幅

// ターゲットの方向に向き直す
aimAtTarget()

// 毎秒指定の度数で追尾(ホーミング)
homing(90)

// 画面外の壁で跳ね返る
bounce()

// 速度の乗算変化 (倍率, 遅延秒数)
fast(2.0, 0.5)  // 0.5秒後に2倍速
slow(0.5, 0.2)  // 0.2秒後に半減

// レーザー予告線と実体化の設定
warningTime = 1.0  // 予告1秒
activeTime = 1.5   // 実体化1.5秒
laserWidth = 12    // 幅12px
組み込み変数と衝突判定
// --- 共通変数 ---
angle, speed   // 角度、速度
spriteAngle    // 画像の回転角度 (見た目の向きのみ)
x, y           // 現在のX, Y座標
xy             // 現在のX, Y座標ペア (例: xy = "100,200" で一括同期)
tx, ty         // ターゲット(相手)のX, Y座標
player_x, player_y // プレイヤーの座標(絶対座標)
enemyHp, enemyMaxHp // 敵ボスの現在HP、最大HP
dist           // 相手との直線距離
second, timer  // 経過秒数 (弾は発射から、コアはカード開始から)
frame          // 経過フレーム数
cardSecond     // 【弾側】カード開始からの秒数
cardFrame      // 【弾側】カード開始からのフレーム数

// --- 弾側専用の変数 ---
ex, ey         // 発射された瞬間（初期位置）のエミッターの位置
e_ex, e_ey     // コア側エミッターの「最新のリアルタイム位置」（追従用）
e_変数名       // コア側のカスタム変数 (例: e_xspa)
color          // 弾の色コード
laserWidth     // レーザーの現在の太さ
hitRadius      // 弾の当たり判定半径 (px)
multf          // 弾の前後倍率 (初期値: 1、後端基準で前方に伸縮。旧: asba)
multlr         // 弾の左右倍率 (初期値: 1、左右に伸縮。旧: aslr)
hitmultf       // 当たり判定専用の前後倍率 (未設定時は multf と連動)
hitmultlr      // 当たり判定専用の左右倍率 (未設定時は multlr と連動)
auraRange      // 【光弾専用】オーラの範囲倍率 (初期値: 2.75)
auraIntensity  // 【光弾専用】オーラの強さ倍率 (初期値: 1.0)

// --- 条件用特殊変数 ---
n              // 自然数 (1, 2, 3...) (例: cardSecond == 5 * n で5秒おきに判定)
               // ※ 誤差許容は == を使用し、厳密一致は === を使用します。

// --- 衝突・状態判定 (0 または 1) ---
isBounced      // bounce() で壁反射した瞬間
isTouchWall    // 壁に衝突した瞬間 (画面の枠線10px手前)
touchingWall   // 壁に接触中
leftWall, rightWall, topWall, bottomWall // 各壁の接触状態
isTouchEdge    // 完全に画面外枠（画面端）に達した瞬間
touchingEdge   // 画面端に接触中（画面外にいる）
leftEdge, rightEdge, topEdge, bottomEdge // 各画面端の接触状態
isTouchBullet  // 他の弾に衝突した瞬間
touchingBullet // 他の弾に接触中
touchColor     // 接触した弾の色
touchX, touchY // 接触した弾の位置
パラメータの選択肢 & 組み込み関数
// 【画像の種類】
"none" (丸ドロー), "light" (光弾), "sword" (剣弾), "normal" (通常弾画像), "marutama" (丸弾), "kome" (米弾), "ootama" (大玉), 
"poihuru" (ポイフル), "star" (星), "uroko" (鱗), "virus" (ウイルス), "onmyoutama" (陰陽玉), "knife" (ナイフ), "ohuda" (お札),
"pallets/色+形状" (パレット弾全165種: 例 "rednormal", "aquastar", "goldbig" 等。色相変色を受けず元の色のまま描画)

// 【弾の種類 (bulletType)】
"normal" (通常弾), "laser" (レーザー), "palette" (パレット弾: colorは内部パラメータとしてのみ機能), "trail" (軌跡弾)

// 【座標モード】
"relative" // 発射台の動きに合わせて弾も一緒に動く (相対)
"absolute" // 発射後は発射台が動いても弾は独立して進む (絶対)

// 【組み込み数学関数】
random()        // 0.0 以上 1.0 未満の乱数
random(max)     // 0 から max 未満の乱数
random(min,max) // min から max 未満の乱数
rand()          // random の略称 (同じ動作)
seedrandom[seed](min,max) // 再現性のあるハッシュブレンド式乱数
sin(度)         // 正弦 (サイン, 度数法)
cos(度)         // 余弦 (コサイン, 度数法)
tan(度)         // 正接 (タンジェント, 度数法)
sqrt(値)        // 平方根
abs(値)         // 絶対値
min(a, b)       // 小さい方の値
max(a, b)       // 大きい方の値
PI, PI2         // 円周率 (3.1415...), 2倍円周率 (6.283...)
